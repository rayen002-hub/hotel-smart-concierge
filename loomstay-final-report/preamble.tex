% --- Figure sécurisée : affiche l'image si elle existe, sinon un placeholder ---
\newcommand{\safefig}[4][0.9]{%
  % #1 = width (default 0.9\textwidth)
  % #2 = path relative to graphicspath (e.g. diagrams/sprint1_usecase.png)
  % #3 = caption
  % #4 = label
  \begin{figure}[H]
    \centering
    \IfFileExists{imgs/#2}{%
      \includegraphics[width=#1\textwidth]{#2}%
    }{%
      \fbox{\begin{minipage}[c][4cm][c]{0.88\textwidth}%
        \centering\textcolor{codegray}{\textit{Image à ajouter :}}\\[0.3cm]%
        \texttt{imgs/#2}%
      \end{minipage}}%
    }
    \caption{#3}
    \label{fig:#4}
  \end{figure}
  \FloatBarrier
}

% --- Screenshot placeholder ---

  % #2 = caption

  \safefig{#1}{#2}{#3}
}
